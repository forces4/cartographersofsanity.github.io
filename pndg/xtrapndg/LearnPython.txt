import os

TEMPLATE = open("template.html", "r", encoding="utf-8").read()

for filename in os.listdir("texts"):
    if filename.endswith(".txt"):
        content = open(f"texts/{filename}", "r", encoding="utf-8").read()
        page = TEMPLATE.replace("<!-- INSERT_CONTENT_HERE -->", content)
        outname = filename.replace(".txt", ".html")
        open(f"output/{outname}", "w", encoding="utf-8").write(page)
